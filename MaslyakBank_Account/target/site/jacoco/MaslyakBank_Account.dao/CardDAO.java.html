package MaslyakBank_Account.dao;

import entity.CardTable;
import lombok.AllArgsConstructor;
import org.hibernate.Session;
import org.hibernate.SessionFactory;
import org.hibernate.Transaction;
import org.springframework.stereotype.Repository;

@Repository
@AllArgsConstructor
public class CardDAO {

    private final SessionFactory sessionFactory;


    public CardTable saveCard(CardTable card) {
        Session session = null;
        Transaction transaction = null;
        try {
            session = sessionFactory.openSession();
            transaction = session.beginTransaction();
            session.merge(card);
            transaction.commit();
            return card;
        } catch (Exception e) {
            if (transaction != null) {
                transaction.rollback();
            }
            throw e;
        } finally {
            if (session != null) {
                session.close();
            }
        }
    }

    public CardTable getCardByNumber(String cardNumber) {
        Session session = null;
        try {
            session = sessionFactory.openSession();
            return session.createQuery("FROM CardTable WHERE cardNumber = :cardNumber", CardTable.class)
                    .setParameter("cardNumber", cardNumber)
                    .uniqueResult();
        } finally {
            if (session != null) {
                session.close();
            }
        }
    }
}
